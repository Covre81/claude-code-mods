# Cofre Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Mod `cofre` que impede segredo (chave de API, token, JWT) de passar pela conversa com o modelo — colado, lido, impresso ou commitado.

**Architecture:** Regras puras em `cofre/hooks/secrets.ts` (detector, nomes de arquivo, `.env`, máscara), ganchos em `cofre/hooks/register.tsx` (`tool.call`, `prompt.mention`, `prompt.submit`, `/chave`). O alerta de commit vira o risco `secret-commit` no `guarda`, que já segura `git` em `tool.call`. A máscara edita o resultado que `next(e)` devolveu em vez de forjar um, porque `BuiltinToolResults` é vazio no `index.d.ts` e o formato de cada ferramenta só é validado em execução.

**Tech Stack:** TypeScript/TSX na API de mods do Claude Code (`claude-code`, `claude-code/testing`), testes com `claude plugin test <dir>`.

**Spec:** `docs/superpowers/specs/2026-10-13-cofre-design.md`

**Regras para quem executa:**
- Só valores **fictícios** nos testes, montados com `'x'.repeat(n)`: assim nenhum literal parece chave real para o GitHub push protection.
- Rodar a partir de `%USERPROFILE%\.claude\mods`.
- Commits terminam com `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```ts
// Pure rules of the cofre: what a secret looks like, which files hold one, and how to hide it.

export type Hit = { kind: string; sample: string }

// ponytail: fixed provider prefixes, no entropy scoring (false positives on hashes and UUIDs); add a row when a new provider shows up.
const PATTERNS: readonly (readonly [string, RegExp])[] = [
  ['anthropic', /sk-ant-[\w-]{20,}/g],
  ['openai', /sk-(?!ant-)(?:proj-)?[\w-]{20,}/g],
  ['google', /AIza[\w-]{35}/g],
  ['jwt', /eyJ[\w-]{10,}\.eyJ[\w-]{10,}\.[\w-]{10,}/g],
  ['github', /ghp_[A-Za-z0-9]{36}|github_pat_\w{22,}/g],
  ['aws', /AKIA[0-9A-Z]{16}/g],
  // The whole block when its END is there, so masking hides the key body, not just the header.
  ['private-key', /-----BEGIN [A-Z ]*PRIVATE KEY-----(?:[\s\S]*?-----END [A-Z ]*PRIVATE KEY-----)?/g],
]

export function detect(text: string): Hit[] {
  return PATTERNS.flatMap(([kind, re]) => [...text.matchAll(re)].map(m => ({ kind, sample: `${m[0].slice(0, 6)}…` })))
}

const baseName = (path: string) => (path.replace(/["']/g, '').split(/[\\/]/).pop() ?? '').toLowerCase()

export const isEnvFile = (path: string) => /^\.env(\..+)?$/.test(baseName(path)) && !/\.(example|sample|template)$/.test(baseName(path))

export function isSecretFile(path: string): boolean {
  const base = baseName(path)
  if (isEnvFile(path)) return true
  if (base.endsWith('.pub')) return false
  return /\.(pem|key)$/.test(base) || /^credentials.*\.json$/.test(base) || /^id_(rsa|ed25519)/.test(base)
}
```

```ts
export const MASK = '••••'

const ENV_LINE = /^\s*(?:export\s+)?([A-Za-z_]\w*)\s*=\s*(.*)$/

function unquote(raw: string): string {
  const v = raw.trim()
  const q = /^(["'])(.*)\1$/.exec(v)
  if (q) return q[1] === '"' ? q[2]!.replace(/\\(["\\])/g, '$1') : q[2]!
  return v.replace(/\s+#.*$/, '')
}

const envEntries = (text: string) =>
  text.split(/\r?\n/).flatMap(line => {
    const m = ENV_LINE.exec(line)
    return m ? [{ name: m[1]!, value: unquote(m[2]!) }] : []
  })

export const envNames = (text: string) => envEntries(text).map(e => e.name)

/** Values worth masking: 8+ chars, so PORT=3000 or DEBUG=true stay readable. */
export const envValues = (text: string) => envEntries(text).map(e => e.value).filter(v => v.length >= 8)

export function maskText(text: string, values: readonly string[]): string {
  let out = text
  for (const v of [...values].sort((a, b) => b.length - a.length)) out = out.split(v).join(MASK)
  for (const [, re] of PATTERNS) out = out.replace(re, MASK)
  return out
}

/** Masks every string inside a tool's result and keeps its shape, so core's schema check still passes. */
export function maskDeep<T>(value: T, values: readonly string[]): T {
  if (typeof value === 'string') return maskText(value, values) as T
  if (Array.isArray(value)) return value.map(v => maskDeep(v, values)) as T
  if (value && typeof value === 'object') return Object.fromEntries(Object.entries(value).map(([k, v]) => [k, maskDeep(v, values)])) as T
  return value
}

const READERS = new Set(['cat', 'type', 'get-content', 'gc', 'more', 'less', 'head', 'tail', 'source', '.'])

/** The secret file a shell command reads, or null. */
export function secretReadTarget(command: string): string | null {
  // ponytail: plain token split, not the guarda's classify; the output mask is the real net.
  const tokens = command
    .split(/[\s;|&()<>]+/)
    .map(t => t.replace(/^["']|["']$/g, ''))
    .filter(Boolean)
  if (!tokens.some(t => READERS.has(t.toLowerCase()))) return null
  return tokens.find(isSecretFile) ?? null
}

/** `.env` text with NAME set to value; double quotes when the value has a space, #, quote or backslash. */
export function upsertEnv(env: string, name: string, value: string): string {
  const line = `${name}=${/[\s#"'\\]/.test(value) ? `"${value.replace(/[\\"]/g, c => `\\${c}`)}"` : value}`
  const re = new RegExp(`^\\s*(?:export\\s+)?${name}\\s*=.*$`, 'm')
  if (re.test(env)) return env.replace(re, () => line)
  return `${env}${env && !env.endsWith('\n') ? '\n' : ''}${line}\n`
}

const COVERS_ENV = new Set(['.env', '/.env', '.env*', '/.env*', '*.env'])

export function ensureGitignore(text: string): string {
  if (text.split(/\r?\n/).some(l => COVERS_ENV.has(l.trim()))) return text
  return `${text}${text && !text.endsWith('\n') ? '\n' : ''}.env\n`
}
```

`name` em `upsertEnv` é seguro dentro do `RegExp` porque o `/chave` só aceita `^[A-Z_][A-Z0-9_]*$` (Task 5).

```tsx
import type { EngineInterface, Register } from 'claude-code'

import { envNames, envValues, isEnvFile, isSecretFile, maskDeep, secretReadTarget } from './secrets'

type Call = Record<string, unknown> & { tool?: unknown }
const str = (v: unknown) => (typeof v === 'string' ? v : '')

/** Values of the session folder's .env files, plus the one being read. ponytail: reread per call, no cache; .env files are tiny. */
async function secretValues($: EngineInterface, extra?: string): Promise<string[]> {
  const local = (await $.fs.list().catch(() => [])).filter(f => f.kind === 'file' && isEnvFile(f.name)).map(f => f.name)
  const paths = extra && !local.includes(extra) ? [...local, extra] : local
  const texts = await Promise.all(paths.map(p => $.fs.read(p).catch(() => '')))
  return texts.flatMap(t => envValues(String(t)))
}

/** Why this call must not run, or null. A Read of .env runs and gets masked instead. */
async function refusal($: EngineInterface, call: Call): Promise<string | null> {
  const tool = str(call.tool)
  const target =
    tool === 'Read' ? str(call.file_path) : tool === 'Grep' ? ([str(call.path), str(call.glob)].find(isSecretFile) ?? '') : (secretReadTarget(str(call.command)) ?? '')
  if (!target || !isSecretFile(target)) return null
  if (tool === 'Read' && isEnvFile(target)) return null
  const names = isEnvFile(target) ? envNames(String(await $.fs.read(target).catch(() => ''))) : []
  return `Arquivo de segredo (${target}): conteúdo oculto pelo cofre.${names.length ? ` Variáveis definidas: ${names.join(', ')}.` : ''} Use os nomes; os valores ficam só no disco.`
}

export const register: Register = on => {
  on('tool.call', { tool: ['Read', 'Grep', 'Bash', 'PowerShell'] }, async ($, e, next) => {
    const call = e as unknown as Call
    const why = await refusal($, call)
    if (why) return { deny: why }
    const r = await next(e)
    if (r.deny !== undefined || r.result === undefined) return r
    const extra = str(call.tool) === 'Read' && isEnvFile(str(call.file_path)) ? str(call.file_path) : undefined
    const masked = maskDeep(r.result, await secretValues($, extra))
    // A hook's own result carries no `text`/`ref`: spreading r would hand the model the raw text core rendered.
    return JSON.stringify(masked) === JSON.stringify(r.result) ? r : ({ result: masked, context: r.context, isError: r.isError } as typeof r)
  }).catch(() => ({ deny: 'O cofre falhou ao conferir esta chamada e reteve o resultado por segurança.' }))

  on('prompt.mention', ($, e, next) => (isSecretFile(e.path) ? { deny: 'arquivo de segredo: o cofre não anexa o conteúdo' } : next(e))).catch(() => ({
    deny: 'o cofre falhou ao conferir a menção',
  }))
}
```

```ts
/** Each simple command of a shell line with the folder it runs in (after any cd). */
function* commands(command: string): Generator<{ cmd: string; args: string[]; dir: Dir }> {
  let dir: Dir = null
  for (const seg of command.split(/&&|\|\||;|\||\n/)) {
    const words = tokenize(seg.trim().replace(/^[({]+\s*/, '').replace(/\s*[)}]+$/, ''))
    while (words.length && (PREFIXES.has(words[0]!) || /^[A-Za-z_]\w*=/.test(words[0]!))) words.shift()
    const [first, ...args] = words
    if (!first) continue
    const cmd = first.replace(/^\\/, '').replace(/\.exe$/i, '')
    if (cmd === 'cd') {
      dir = joinDir(dir, args[0])
      continue
    }
    yield { cmd, args, dir }
  }
}

/** The first risky segment of a shell command, or null. A bare `npm install` / `pip install -r` adds nothing new and passes. */
export function classify(command: string): Risk | null {
  for (const { cmd, args, dir } of commands(command)) {
    const risk = classifyCmd(cmd, args, dir)
    if (risk) return risk
  }
  return null
}

function gitParts(args: string[], dir: Dir) {
  let i = 0
  let gitDir = dir
  while (i < args.length && args[i]!.startsWith('-')) {
    if (args[i] === '-C') gitDir = joinDir(gitDir, args[i + 1])
    i += args[i] === '-C' || args[i] === '-c' ? 2 : 1
  }
  return { sub: args[i], rest: args.slice(i + 1), gitDir }
}

/** A git add/commit that may carry a secret file; only git itself can tell, so register.tsx asks it before holding. */
export function commitCandidate(command: string): Extract<Risk, { kind: 'secret-commit' }> | null {
  for (const { cmd, args, dir } of commands(command)) {
    if (cmd !== 'git') continue
    const { sub, rest, gitDir } = gitParts(args, dir)
    if (sub === 'add' || sub === 'commit') return { kind: 'secret-commit', label: `git ${sub}`, args: rest, dir: gitDir }
  }
  return null
}
```

```ts
/** Secret files this add/commit would put in history: `git add -n` for add, the index (+ worktree with -a) for commit. */
async function stagedSecrets($: EngineInterface, risk: Extract<Risk, { kind: 'secret-commit' }>, cwd: string): Promise<string[]> {

    risk.label === 'git add'
      ? (await out(['add', '-n', ...risk.args])).map(l => /^add '(.+)'$/.exec(l.trim())?.[1] ?? '')

          ...(await out(['diff', '--cached', '--name-only'])),
          ...(risk.args.includes('--all') || risk.args.some(a => /^-[^-]*a/.test(a)) ? await out(['diff', '--name-only']) : []),
        ]
  return [...new Set(files.map(f => f.trim()).filter(f => f && isSecretFile(f)))]
}
```
